## Review Focus

1. A flag write fails halfway through an engine change while the user chose "restart now" → NVDA must not save or restart with a half-written engine set (Task 6, `test_engine_change_does_not_restart_after_failed_flag_write`).
2. A `wx.Choice` ("options") field fails during panel save → earlier fields stay written, later fields keep their old values, nothing is reset to the first option (Task 6, `test_failed_choice_save_keeps_unwritten_settings`).

### Task 3: Translator comments follow NVDA's rule

```python
"""Translator comments must sit directly above a gettext call (NVDA codingStandards.md)."""

import re
from pathlib import Path


ADDON = Path(__file__).resolve().parents[1] / "addon"
# Same exclusions as pyproject.toml and .pre-commit-config.yaml.
EXCLUDED = re.compile(r"^synthDrivers/WorldVoice/(driver/[^/]+/driver/|languageDetection/script_data\.py$)")
GETTEXT_CALL = re.compile(r"\b(_|pgettext|ngettext|npgettext)\(")
# NVDA's multi-line form places the comment inside the call's parentheses.
GETTEXT_OPEN = re.compile(r"\b(_|pgettext|ngettext|npgettext)\($")


def _misplaced_lines(lines):
	"""Return 1-based line numbers of translator comments not followed by a gettext call."""
	misplaced = []
	for number, line in enumerate(lines):
		if not line.lstrip().startswith("# Translators:"):
			continue
		if number > 0 and GETTEXT_OPEN.search(lines[number - 1].rstrip()):
			continue
		following = number + 1
		while following < len(lines) and lines[following].lstrip().startswith("#"):
			following += 1
		if following == len(lines) or not GETTEXT_CALL.search(lines[following]):
			misplaced.append(number + 1)
	return misplaced


def test_scanner_accepts_nvda_multiline_comment_and_rejects_separated_comment():
	nvda_example = [
		"self.copySettingsButton = wx.Button(",
		"	self,",
		"	label=_(",
		"		# Translators: The label for a button in general settings to copy",
		"		# current user settings to system settings.",
		'		"Use currently saved settings during sign-in and on secure screens"',
		"	)",
		")",
		"# Translators: The name of a category of NVDA commands.",
		'SCRCAT_TEXTREVIEW = _("Text review")',
	]
	assert _misplaced_lines(nvda_example) == []
	assert _misplaced_lines(["# Translators: Title.", "gui.mainFrame,", 'message=_("Export")']) == [1]
	assert _misplaced_lines(["# Translators: Dangling at end of file."]) == [1]


def test_translator_comments_precede_gettext_calls():
	misplaced = []
	for path in sorted(ADDON.rglob("*.py")):
		relative = path.relative_to(ADDON).as_posix()
		if EXCLUDED.search(relative):
			continue
		lines = path.read_text(encoding="utf-8").splitlines()
		misplaced.extend(f"{relative}:{number}" for number in _misplaced_lines(lines))
	assert misplaced == []
```

### Task 4: C1 – eSpeak and RHVoice fail segments after terminate

Today both wrappers `return` silently when `speak_segment` arrives after `terminate()`, so the scheduler waits forever for a terminal event. OneCore, SAPI5 and VE already report `SEGMENT_FAILED`. The fix routes the rejection through `SegmentLifecycle.reject_new`. Its docstring says it delivers "even after close", on the lifecycle delivery thread rather than the caller's thread.

Segments that were already accepted and then terminated must stay silent. `test_cancel_or_terminate_before_deferred_delivery_keeps_event_silent` in the eSpeak tests already pins that; do not change it.

```python
import ast
from collections import OrderedDict
from pathlib import Path
import types

import pytest


PANEL_PATH = Path(__file__).resolve().parents[1] / "addon/globalPlugins/WorldVoice/speechSettingsDialog.py"


class RecordingSection(dict):
	def __init__(self, calls, values):
		super().__init__(values)
		self._calls = calls

	def __setitem__(self, key, value):
		self._calls.append(("write", key, value))
		super().__setitem__(key, value)


def _load_panels(calls, logged):
```
