## Review Focus

1. **A player stop failure during NVDA's own `_stopThread()`** (voice reload or terminate): the stop now latches before raising. The reload must still leave the driver recoverable, because `_initWasapiAudio` calls `host.reset()`. Test in Task 3.
2. **NVDA bookmark index callbacks** (`feed(None, 0, onDone)`) firing after `EndStream` closed admission: they must still reach `onIndexReached` while the record is valid. The SAPI5 `_callback` hook keeps this. The existing ordinary-notification tests cover it; confirm they still pass and name them in the ledger.
3. **`cancel()` racing the worker**: the stop gate is now raised before admission is taken. The worker still never pops a request while a cancel's stop runs. The existing `test_cancelled_ordinary_pcm_feed_stops_again_before_successor_admission` and the cancel tests cover it; run them repeatedly (`-p no:randomly --count` is unavailable, so loop 20× in shell).
4. **A fault raised by the worker before `__init__` returns** (during the first `_initWasapiAudio`): the policy and gate must already exist. Create them before the worker starts. Test in Task 3.
5. **A `ControlFaultPolicy.fail` reached while `_threadCond` is held by the caller** (e.g. a stop inside `stop_native`'s locked section): `abandon` must not run under that lock. The policy releases the condition before abandoning. But a caller that already holds the RLock keeps it, so audit every call site. Task 3 lists them.

```python
"""Parts an engine wrapper uses around its own native queue and worker.

Each part is guarded by the engine's own condition (its queue lock), which
comes after host admission and before the host condition in the lock order.
"""

from collections.abc import Callable, Hashable, Iterator
from contextlib import contextmanager
from threading import Condition

from .segmentHost import NativeRecord, SegmentHost


class StopGate:
	"""Stops in flight on an engine's queue; its worker takes no request until none remain."""

	def __init__(self, condition: Condition):
		self._condition = condition
		self._stops = 0

	@contextmanager
	def hold(self) -> Iterator[None]:
		with self._condition:
			self._stops += 1
		try:
			yield
		finally:
			with self._condition:
				self._stops -= 1
				self._condition.notify_all()

	@property
	def clear(self) -> bool:
		"""Whether no stop is in flight; read it while holding the condition."""
		return self._stops == 0


class QueuedRecords:
	"""Records submitted to an engine queue and not yet taken by its worker.

	The caller guards it with the lock that also guards the engine's queue.
	"""

	def __init__(self):
		self._records: dict[Hashable, NativeRecord] = {}

	def add(self, key: Hashable, record: NativeRecord) -> None:
		self._records[key] = record

	def pop(self, key: Hashable) -> NativeRecord | None:
		return self._records.pop(key, None)

	def take_all(self) -> list[NativeRecord]:
		records = list(self._records.values())
		self._records.clear()
		return records

	def __len__(self) -> int:
		return len(self._records)


class ControlFaultPolicy:
	"""Latch the first control error, drop queued work and wake the engine's worker."""

	def __init__(
		self,
		host: SegmentHost,
		condition: Condition,
		drop: Callable[[], list[NativeRecord]],
	):
		self._host = host
		self._condition = condition
		self._drop = drop

	def fail(self, error: Exception) -> bool:
		"""Return False when a control error was already latched."""
		host = self._host
		with self._condition:
			if host.control_error is not None:
				return False
			host.latch_control_error(error)
			dropped = self._drop()
			self._condition.notify_all()
		# Queued work never reaches the worker; it fails with the latched error.
		# abandon() waits for consumers, which may need the engine lock.
		for record in dropped:
			host.abandon(record)
		return True
```

```python
class LeasedPlayer:
	"""Admit every feed/idle as a consumer of the current record.

	Engines adjust three hooks: how a feed's onDone callback is wrapped, how a
	feed error is recorded, and what a failed stop does.
	"""

	def __init__(...)  # unchanged

	def __getattr__(...)  # unchanged

	def feed(self, data, size=None, onDone=None):
		host = self._host
		with host.consumer() as record:
			if record is None:
				return None
			callback = None if onDone is None else self._callback(record, onDone)
			try:
				if size is None:
					return self._player.feed(data, onDone=callback)
				return self._player.feed(data, size, onDone=callback)
			except Exception as error:
				self._record_error(record, error)
				raise
			finally:
				# A feed that was already admitted may queue audio after cancel's
				# stop. Stop it again before the FIFO barrier permits a successor.
				if not record.is_valid():
					self.stop()

	def _callback(self, record, onDone):
		host = self._host

		def callback():
			with host.consumer(record) as admitted:
				if admitted is None:
					return
				with host.bound_callback(record):
					try:
						onDone()
					except Exception as error:
						host.record_error(record, error)
						raise

		return callback

	def _record_error(self, record, error):
		self._host.record_error(record, error)

	def _stop_failed(self, error):
		self._host.latch_control_error(error)

	# idle unchanged

	def stop(self):
		try:
			return self._player.stop()
		except Exception as error:
			self._stop_failed(error)
			raise
```

class _SegmentPlayer(LeasedPlayer):
	"""SAPI5's WavePlayer lease: legacy feeds pass through, index callbacks outlive admission."""

	def __init__(self, driver, player):
		super().__init__(driver._segmentHost, player, idle_barrier=None)
		self._driver = weakref.ref(driver)

	def feed(self, data, size=None, onDone=None):
		driver = self._driver()
		if driver is None:
			return None
		with driver._threadCond:
			if self is not driver.player:
				return None
		if driver._segmentHost.current() is None:
			return self._player.feed(data, size, onDone)
		# A feed nested in an admitted PCM callback may pass after EndStream
		# closed admission; any other late feed is dropped.
		return super().feed(data, size, onDone)

	def _callback(self, record, onDone):
		# NVDA queues bookmark indexes as feed(None, 0, onDone); they fire after
		# EndStream closed admission, while the record is still valid.
		def callback():
			if record.is_valid():
				onDone()

		return callback

	def _record_error(self, record, error):
		driver = self._driver()
		if driver is None:
			return super()._record_error(record, error)
		driver._recordSegmentError(record, error)

	def _stop_failed(self, error):
		driver = self._driver()
		if driver is None:
			return super()._stop_failed(error)
		driver._failSegmentControl(error)

		...  # the existing idle body, unchanged

def _sapi5_tree():
	return ast.parse((ROOT / "driver/SAPI5/__init__.py").read_text(encoding="utf-8"))


def _class_constant(tree, class_name, constant):
	node = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == class_name)
	assign = next(
		n for n in node.body
		if isinstance(n, ast.Assign) and any(getattr(t, "id", None) == constant for t in n.targets)
	)
	return set(ast.literal_eval(assign.value.args[0]))


def test_sapi5_nvda_privates_are_listed_and_exist_in_nvda():
	tree = _sapi5_tree()
	privates = _class_constant(tree, "_Sapi5Port", "NVDA_PRIVATES")
	overrides = _class_constant(tree, "_Sapi5Port", "NVDA_OVERRIDES")
	own = set()
	for node in tree.body:
		if isinstance(node, ast.ClassDef):
			for item in node.body:
				if isinstance(item, ast.FunctionDef):
					own.add(item.name)
					if item.name == "__init__":
						own |= {
							n.attr for n in ast.walk(item)
							if isinstance(n, ast.Attribute) and isinstance(n.ctx, ast.Store)
							and isinstance(n.value, ast.Name) and n.value.id == "self"

	accessed = {
